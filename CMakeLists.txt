cmake_minimum_required(VERSION 3.22)
project(DrumBus VERSION 0.12.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(DrumBus COMPANY_NAME LM18 BUNDLE_ID com.lm18.drumbus PLUGIN_MANUFACTURER_CODE Lm18 PLUGIN_CODE Drb1 FORMATS VST3 PRODUCT_NAME "LM18 DrumBus")
juce_add_binary_data(DrumBusReference SOURCES REFNoise1.wav LexRoom1.wav)
target_sources(DrumBus PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_link_libraries(DrumBus PRIVATE DrumBusReference juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
target_compile_definitions(DrumBus PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
